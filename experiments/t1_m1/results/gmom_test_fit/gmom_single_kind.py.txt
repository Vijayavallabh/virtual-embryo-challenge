def rms(f, c):
    cf = f.categories[c]
    free = ~np.isin(np.asarray(f.genes, dtype=object), np.asarray(sorted(held), dtype=object))
    return float(np.sqrt(np.average((cf.logf[:, free] ** 2).mean(axis=1), weights=cf.sizes)))


def lib_slope(f, c):
    """Per gene, the cell-weighted regression slope of ALL's metacell factors on the metacell's library-3 share."""
    cf = f.categories[c]
    M = cf.n_metacells
    s, n = np.zeros(M), np.zeros(M)
    for name, m in cf.index.items():
        n[m] += 1
        s[m] += str(name).rsplit("_", 1)[-1] == "3"
    s = s / np.maximum(n, 1)
    w = np.asarray(cf.sizes, float)
    x = s - np.average(s, weights=w)
    return (w[:, None] * x[:, None] * cf.logf).sum(axis=0) / float((w * x * x).sum())
