"""Target-free listing (decision log 2026-11-18 (g)): can a library-linked component like the one in the test pair's
g_mom factors be emulated inside the practice fit by thinning one library of its reference? The practice fit (start
E8.5, reference E9.5) is refitted with the counts of the reference's cells from one library thinned to 2/3 and to 1/3
(Binomial thinning, one fixed stream), and the share of each category's factors' squared size that goes with a
metacell's share of that library is read with its permutation null, beside the size of the factors. Reads the released
E8.5 and E9.5 stages only; builds no prediction, reads no board, no target and no score. Shares and ratios only.
Usage, repository root: uv run --frozen --no-sync python -I -B <this file>"""
import sys
from pathlib import Path

import anndata as ad
import numpy as np
from scipy import sparse

REPO = Path.cwd()
sys.path.insert(0, str(REPO / "experiments" / "t1_m1"))
from vec import core, harness, prep, t1_scope  # noqa: E402
from vec import t1_state_transport as ts  # noqa: E402
from vec.prep import load_stage  # noqa: E402

data = harness.DATA_DIR
e85 = load_stage(data / "T1/E8.5_RNA.h5ad", lineage=False)
e95 = load_stage(data / "T1/E9.5_RNA.h5ad", lineage=False)
held = t1_scope.held_flat_genes()
cats = [c for c, s in harness.t1_chamber_shares(e95, "E9.5_RNA").items() if s > 0]
libs = np.array([str(n).rsplit("_", 1)[-1] for n in e95.obs_names])
print("reference E9.5: cells by library " + ", ".join(f"{b}: {float((libs == b).mean()):.2f}" for b in sorted(set(libs))))
C0 = sparse.csr_matrix(core.counts_of(e95))
assert float(abs(prep.log1p_cp10k(C0) - e95.X).max()) < 1e-5


def thinned(lib: str, p: float, stream: int) -> ad.AnnData:
    rng = np.random.default_rng([20261118, stream])
    C = C0.copy().astype(np.int64)
    rows = np.flatnonzero(libs == lib)
    for r in rows:
        a, b = C.indptr[r], C.indptr[r + 1]
        C.data[a:b] = rng.binomial(C.data[a:b], p)
    C.eliminate_zeros()
    out = ad.AnnData(X=prep.log1p_cp10k(C), obs=e95.obs.copy(), var=e95.var.copy(), obsm=dict(e95.obsm), uns=dict(e95.uns))
    out.layers["counts"] = sparse.csr_matrix(C, dtype=np.int32)
    return out


def reading(ref: ad.AnnData, label: str, lib: str) -> dict:
    fin, raw = harness.g_mom_pair_factors(ref, "E9.5_RNA", e85, "E8.5_RNA", categories=cats)
    free = ~np.isin(np.asarray(raw.genes, dtype=object), np.asarray(list(held), dtype=object))
    rng = np.random.default_rng(7)
    out = {}
    for c in sorted(fin.categories):
        cf = fin.categories[c]
        M = cf.n_metacells
        s, n = np.zeros(M), np.zeros(M)
        for name, m in cf.index.items():
            n[m] += 1
            s[m] += str(name).rsplit("_", 1)[-1] == lib
        s = s / np.maximum(n, 1)
        w = np.asarray(cf.sizes, float)
        Y = cf.logf[:, free]
        tot = float((w[:, None] * Y ** 2).sum())

        def r2(sv):
            x = sv - np.average(sv, weights=w)
            den = float((w * x * x).sum())
            if den == 0 or tot == 0:
                return float("nan")
            beta = (w[:, None] * x[:, None] * Y).sum(axis=0) / den
            return float((den * beta ** 2).sum() / tot)

        null = [r2(rng.permutation(s)) for _ in range(200)]
        rms = float(np.sqrt(np.average((Y ** 2).mean(axis=1), weights=w)))
        out[c] = (r2(s), float(np.mean(null)), float(np.percentile(null, 95)), rms, M)
        print(f"  {label} {c}: {M} metacells; share of the finished factors' squared size that goes with the metacell's share of "
              f"library {lib}: {out[c][0]:.3f} (permutation null mean {out[c][1]:.3f}, 95th percentile {out[c][2]:.3f}); rms {rms:.4f}")
    return out


minority = min(sorted(set(libs)), key=lambda b: float((libs == b).mean()))
print(f"the library thinned: {minority} (the smaller one, as the shallower libraries are the smaller part of E10.5)")
base = reading(e95, "as released", minority)
for p, stream in ((2 / 3, 1), (1 / 3, 2)):
    ref = thinned(minority, p, stream)
    got = reading(ref, f"library {minority} thinned to {p:.2f}", minority)
    for c in got:
        print(f"     {c}: rms against the released fit x{got[c][3] / base[c][3]:.2f}")
